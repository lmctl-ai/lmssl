cmake_minimum_required(VERSION 3.24)
project(lmssl VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(LMSSL_FULL_STATIC "Link the complete executable statically" ON)

# ---- schema embedded as bytes: the binary prints exactly what the compiler hashed
set(LMSSL_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmssl.dsl.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LMSSL_SCHEMA}")
file(READ "${LMSSL_SCHEMA}" LMSSL_SCHEMA_HEX HEX)
string(LENGTH "${LMSSL_SCHEMA_HEX}" LMSSL_SCHEMA_HEX_LENGTH)
math(EXPR LMSSL_SCHEMA_LAST_BYTE "${LMSSL_SCHEMA_HEX_LENGTH} - 2")
set(LMSSL_SCHEMA_INITIALIZER "")
foreach(offset RANGE 0 ${LMSSL_SCHEMA_LAST_BYTE} 2)
  string(SUBSTRING "${LMSSL_SCHEMA_HEX}" ${offset} 2 byte)
  string(APPEND LMSSL_SCHEMA_INITIALIZER "0x${byte},")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/schema_embedded.hpp"
  "#pragma once\n#include <cstddef>\nnamespace lmssl { inline constexpr unsigned char embedded_schema[] = {${LMSSL_SCHEMA_INITIALIZER}}; inline constexpr std::size_t embedded_schema_size = sizeof(embedded_schema); }\n")

# ---- OpenSSL's stock apps/openssl.cnf, used when the system has none (a static binary has no OPENSSLDIR)
set(LMSSL_STOCK_CNF "${CMAKE_CURRENT_SOURCE_DIR}/third_party/openssl/apps/openssl.cnf")
if(NOT EXISTS "${LMSSL_STOCK_CNF}")
  message(FATAL_ERROR "third_party/openssl missing: run scripts/sync-openssl.sh")
endif()
file(READ "${LMSSL_STOCK_CNF}" LMSSL_CNF_HEX HEX)
string(LENGTH "${LMSSL_CNF_HEX}" LMSSL_CNF_HEX_LENGTH)
math(EXPR LMSSL_CNF_LAST_BYTE "${LMSSL_CNF_HEX_LENGTH} - 2")
set(LMSSL_CNF_INITIALIZER "")
foreach(offset RANGE 0 ${LMSSL_CNF_LAST_BYTE} 2)
  string(SUBSTRING "${LMSSL_CNF_HEX}" ${offset} 2 byte)
  string(APPEND LMSSL_CNF_INITIALIZER "0x${byte},")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/cnf_embedded.hpp"
  "#pragma once\n#include <cstddef>\nnamespace lmssl { inline constexpr unsigned char stock_cnf[] = {${LMSSL_CNF_INITIALIZER}}; inline constexpr std::size_t stock_cnf_size = sizeof(stock_cnf); }\n")

# ---- OpenSSL: built from the pinned source in third_party/openssl (scripts/sync-openssl.sh)
include(cmake/openssl.cmake)

add_executable(lmssl cpp/main.cpp cpp/host.cpp)
target_compile_definitions(lmssl PRIVATE LMSSL_VERSION="${PROJECT_VERSION}")
target_include_directories(lmssl PRIVATE cpp "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(lmssl PRIVATE lmssl_openssl_apps)
target_compile_options(lmssl PRIVATE -Wall -Wextra)
if((CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang") AND NOT APPLE)
  target_link_options(lmssl PRIVATE -static-libstdc++ -static-libgcc)
  if(LMSSL_FULL_STATIC)
    target_link_options(lmssl PRIVATE -static)
  endif()
endif()
